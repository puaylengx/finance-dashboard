beforeEach Guard Logic

// router/index.ts
router.beforeEach((to, _from, next) => {
  const session  = getSession()
  const role     = session?.role ?? ''
  const position = session?.position ?? null

  // ยังไม่ login → /login
  if (to.meta.requiresAuth && !session) return next('/login')

  // role='user' = ไม่มีสิทธิ์ใดเลย (backend ส่งมาเมื่อไม่รู้จัก job_title)
  if (to.meta.requiresAuth && role === 'user') return next('/login')

  // Admin page: ต้อง FA + position
  if (to.meta.requiresFAPosition && (!isFA(role) || !hasPosition(position)))
    return next('/io')

  // login แล้วไปหน้า login → redirect /budget
  if (to.path === '/login' && session) return next('/budget')
})
ℹ Role-based UI rendering (ไม่ใช่ route guard) ทุก role เข้าถึง /budget และ /io ได้ — backend enforce ผ่าน cost_owner filtering Frontend ใช้ auth.isUserFA / auth.isUserDiv เพื่อ toggle ส่วนที่แสดงผล (Division+Dept vs combined)

7API Layer

Axios Client — src/api/client.ts

// Request interceptor — แนบ JWT ทุก request
apiClient.interceptors.request.use(config => {
  const token = getToken()
  if (token) config.headers.Authorization = `Bearer ${token}`
  return config
})

// Response interceptor — 401 → clear session → /login
apiClient.interceptors.response.use(
  res => res,
  err => {
    if (err.response?.status === 401) {
      clearSession()
      window.location.href = '/login'
    }
    return Promise.reject(err)
  },
)

// redirectUri: window.location.origin + '/auth-redirect.html'
// LoginView.vue — Microsoft login handler
async function handleMicrosoftLogin() {
  try {
    const result = await msalInstance.loginPopup({ scopes: loginScopes, redirectUri })
    const res = await entraLogin(result.accessToken)   // POST /auth/entra-login
    auth.setUser(res.data)                             // ตั้ง session + redirect /budget
  } catch (e) {
    // interaction_in_progress error: clear msal.* sessionStorage แล้ว retry ครั้งเดียว
  }
}
// public/auth-redirect.html — MSAL popup redirect handler (ต้องมีไว้)

// fmtDate: long Thai format
function fmtDate(iso: string) {
  const d = new Date(iso)
  const date = d.toLocaleDateString('th-TH', { day: 'numeric', month: 'long', year: 'numeric' })
  const time = d.toLocaleTimeString('th-TH', { hour: '2-digit', minute: '2-digit', hour12: false })
  return `${date} เวลา ${time} น.`
}
